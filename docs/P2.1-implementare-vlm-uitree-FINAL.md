## 2. VLM INTEGRATION

**Transport:** Hydra trimite Base64 direct la LLM endpoint

```python
def vlm_analyze(device_id: str, prompt: str) -> dict:
    """
    1. Ia screenshot resized de pe device
    2. Trimite la LLM cu prompt
    3. Returnează JSON parsat
    """
    # Pas 1: screenshot resized de pe device
    result = phone_agent.send(device_id, "screenshot_for_vlm", {})
    image_base64 = result["image_base64"]
    
    # Pas 2: trimite la LLM (local sau cloud)
    response = llm_client.analyze_image(
        image_base64=image_base64,
        prompt=prompt,
        max_tokens=500
    )
    
    # Pas 3: parsează JSON din răspuns
    return json.loads(response)
```

## 4. SESSION CHECKPOINTS (DECIZIE: PostgreSQL)

```sql
CREATE TABLE execution_checkpoints (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    session_id UUID NOT NULL,
    task_id UUID NOT NULL REFERENCES tasks(id),
    device_id UUID NOT NULL REFERENCES devices(id),
    account_id UUID NOT NULL REFERENCES accounts(id),
    
    phase VARCHAR(50) NOT NULL,  -- 'warm_up', 'search', 'decision_loop', 'cool_down'
    state JSONB NOT NULL,
    /*
    state = {
        "evaluated": 23,
        "matched": 7,
        "last_target": "@username123",
        "scroll_position": 15,
        "partial_results": [...]
    }
    */
    
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW(),
    expires_at TIMESTAMP DEFAULT NOW() + INTERVAL '24 hours',
    
    UNIQUE(task_id, device_id)  -- un checkpoint per task/device
);

CREATE INDEX idx_checkpoints_task ON execution_checkpoints(task_id);
CREATE INDEX idx_checkpoints_expires ON execution_checkpoints(expires_at);
```

```python
CHECKPOINT_SAVE_TRIGGERS = [
    "after_each_match",           # follow/like executat
    "every_5_evaluations",        # chiar dacă no match
    "on_phase_transition",        # warm_up → decision_loop
]

def save_checkpoint(session_id, task_id, device_id, phase, state):
    db.upsert("execution_checkpoints", 
        conflict_on=["task_id", "device_id"],
        values={
            "session_id": session_id,
            "task_id": task_id,
            "device_id": device_id,
            "account_id": state.get("account_id"),
            "phase": phase,
            "state": json.dumps(state),
            "updated_at": now(),
            "expires_at": now() + timedelta(hours=24)
        }
    )

def load_checkpoint(task_id, device_id) -> dict | None:
    row = db.query("""
        SELECT * FROM execution_checkpoints 
        WHERE task_id = ? AND device_id = ? AND expires_at > NOW()
    """, [task_id, device_id])
    return json.loads(row["state"]) if row else None

def clear_checkpoint(task_id, device_id):
    db.delete("execution_checkpoints", 
        where={"task_id": task_id, "device_id": device_id})
```

## 5. SKILL STORAGE (DECIZIE: Hybrid)

```sql
-- Master skill definitions (shared între toate devices)
CREATE TABLE skill_definitions (
    id VARCHAR(50) PRIMARY KEY,  -- 'instagram', 'tiktok'
    platform VARCHAR(50) NOT NULL,
    app_version VARCHAR(20),
    
    selectors JSONB NOT NULL,      -- button_map cu selectors + hints
    navigation_map JSONB NOT NULL, -- ecrane + tranziții
    
    created_at TIMESTAMP DEFAULT NOW(),
    updated_at TIMESTAMP DEFAULT NOW()
);

-- Learned coords per device (cache sync)
CREATE TABLE skill_coords_cache (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    skill_id VARCHAR(50) REFERENCES skill_definitions(id),
    device_id UUID REFERENCES devices(id),
    
    screen_width INT NOT NULL,
    screen_height INT NOT NULL,
    
    coords JSONB NOT NULL,
    /*
    coords = {
        "nav.home": { "x": 0.10, "y": 0.96, "confidence": 0.95, "learned_at": "..." },
        "post.like": { "x": 0.92, "y": 0.45, "confidence": 0.88, "learned_at": "..." }
    }
    */
    
    synced_at TIMESTAMP DEFAULT NOW(),
    
    UNIQUE(skill_id, device_id)
);
```

```python
SYNC_INTERVAL_SEC = 300  # 5 minute

def load_skill(platform: str, device_id: str) -> Skill:
    # 1. Load master definition
    master = db.query("SELECT * FROM skill_definitions WHERE id = ?", [platform])
    
    # 2. Load device-specific coords
    cache = db.query("""
        SELECT coords FROM skill_coords_cache 
        WHERE skill_id = ? AND device_id = ?
    """, [platform, device_id])
    
    # 3. Merge
    skill = Skill(
        selectors=master["selectors"],
        navigation_map=master["navigation_map"],
        learned_coords=cache["coords"] if cache else {}
    )
    return skill

def save_learned_coord(skill_id: str, device_id: str, button_id: str, coord: dict):
    # Update local cache
    db.execute("""
        INSERT INTO skill_coords_cache (skill_id, device_id, screen_width, screen_height, coords)
        VALUES (?, ?, ?, ?, ?::jsonb)
        ON CONFLICT (skill_id, device_id) DO UPDATE SET
            coords = skill_coords_cache.coords || ?::jsonb,
            synced_at = NOW()
    """, [skill_id, device_id, coord["screen_width"], coord["screen_height"],
          json.dumps({button_id: coord}), json.dumps({button_id: coord})])

def sync_coords_to_other_devices(skill_id: str, source_device_id: str):
    """
    Sync learned coords to devices with same resolution.
    Called every 5 min or on session end.
    """
    source = db.query("""
        SELECT screen_width, screen_height, coords 
        FROM skill_coords_cache 
        WHERE skill_id = ? AND device_id = ?
    """, [skill_id, source_device_id])
    
    if not source:
        return
    
    # Find devices with same resolution
    same_res_devices = db.query("""
        SELECT DISTINCT device_id FROM skill_coords_cache
        WHERE skill_id = ? 
        AND device_id != ?
        AND screen_width = ? 
        AND screen_height = ?
    """, [skill_id, source_device_id, source["screen_width"], source["screen_height"]])
    
    for device in same_res_devices:
        # Merge coords, keep highest confidence
        db.execute("""
            UPDATE skill_coords_cache SET
                coords = merge_coords_keep_highest_confidence(coords, ?::jsonb),
                synced_at = NOW()
            WHERE skill_id = ? AND device_id = ?
        """, [json.dumps(source["coords"]), skill_id, device["device_id"]])
```

## 6. RATE LIMITING (LLM Local)

```python
# Concurrency control pentru LLM local
LLM_CONFIG = {
    "max_concurrent": 5,       # max 5 requests paralele
    "queue_timeout_sec": 30,   # așteaptă max 30s în queue
    "retry_on_busy": True,
}

# Semaphore pentru concurrency
llm_semaphore = asyncio.Semaphore(LLM_CONFIG["max_concurrent"])

async def vlm_analyze_with_limit(device_id: str, prompt: str) -> dict:
    try:
        async with asyncio.timeout(LLM_CONFIG["queue_timeout_sec"]):
            async with llm_semaphore:
                return await vlm_analyze(device_id, prompt)
    except asyncio.TimeoutError:
        log.warn(f"LLM queue timeout for device {device_id}")
        return {"error": "llm_queue_timeout", "fallback": True}
```

## 8. FLOW COMPLET: FOLLOW SESSION

```python
async def execute_follow_session(device_id: str, account_id: str, task: dict):
    skill = load_skill("instagram", device_id)
    checkpoint = load_checkpoint(task["id"], device_id)
    
    session_id = checkpoint["session_id"] if checkpoint else str(uuid4())
    
    try:
        # 1. Pregătire device
        await phone_agent.send(device_id, "screen_wake", {})
        await phone_agent.send(device_id, "unlock", {})
        await phone_agent.send(device_id, "open_app", {"packageName": "com.instagram.android"})
        await phone_agent.send(device_id, "wait_for_idle", {"timeoutMs": 5000})
        
        # 2. Resume sau start fresh
        if checkpoint and checkpoint["phase"] == "decision_loop":
            log.info(f"Resuming from checkpoint: {checkpoint['evaluated']} evaluated")
            state = checkpoint
        else:
            # Warm-up
            await warm_up(device_id)
            
            # Navigate to target (search → profile → followers)
            await navigate_to_followers_list(device_id, task["seed"], skill)
            
            state = {
                "session_id": session_id,
                "account_id": account_id,
                "evaluated": 0,
                "matched": 0,
                "results": []
            }
        
        # 3. Decision loop
        results = await decision_loop(device_id, task, skill, state)
        
        # 4. Cool-down
        await tap_element(device_id, "nav.home", skill)
        await cool_down(device_id)
        
        # 5. Close
        await phone_agent.send(device_id, "close_app", {"packageName": "com.instagram.android"})
        await phone_agent.send(device_id, "screen_off", {})
        
        # 6. Clear checkpoint (success)
        clear_checkpoint(task["id"], device_id)
        
        return {"status": "completed", "results": results}
        
    except SoftBlockDetectedError as e:
        save_checkpoint(session_id, task["id"], device_id, "blocked", state)
        raise
        
    except Exception as e:
        save_checkpoint(session_id, task["id"], device_id, "error", state)
        raise
```
